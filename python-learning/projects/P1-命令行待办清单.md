## 一、项目需求

开发一个命令行待办事项工具，支持：

```
todo add 买牛奶              # 添加任务
todo list                    # 列出所有未完成任务
todo done 2                  # 标记第 2 条为完成
todo rm 2                    # 删除第 2 条
todo clear                   # 清空已完成
todo stats                   # 统计信息
```

任务持久化到本地 JSON 文件，程序重启后数据不丢。

### models.py 骨架

```python
from dataclasses import dataclass, field
from datetime import datetime

PRIORITY_ORDER = {"high": 0, "normal": 1, "low": 2}

@dataclass
class Task:
    id: int
    content: str
    priority: str = "normal"
    done: bool = False
    created_at: str = field(default_factory=lambda: datetime.now().isoformat(timespec="seconds"))

    @property
    def sort_key(self) -> tuple[int, str]:
        return (PRIORITY_ORDER[self.priority], self.created_at)

    def to_dict(self) -> dict:
        return {
            "id": self.id, "content": self.content, "priority": self.priority,
            "done": self.done, "created_at": self.created_at,
        }

    @classmethod
    def from_dict(cls, data: dict) -> "Task":
        return cls(
            id=data["id"], content=data["content"],
            priority=data.get("priority", "normal"),
            done=data.get("done", False),
            created_at=data.get("created_at", ""),
        )
```

### storage.py 骨架

```python
import json
import os
from pathlib import Path

class StorageError(Exception): ...

DATA_FILE = Path("tasks.json")

def load() -> list[Task]:
    if not DATA_FILE.exists():
        return []
    try:
        raw = json.loads(DATA_FILE.read_text(encoding="utf-8"))
    except json.JSONDecodeError as e:
        raise StorageError(f"数据文件损坏: {e}") from e
    return [Task.from_dict(d) for d in raw]

def save(tasks: list[Task]) -> None:
    tmp = DATA_FILE.with_suffix(".tmp")
    try:
        tmp.write_text(
            json.dumps([t.to_dict() for t in tasks], ensure_ascii=False, indent=2),
            encoding="utf-8",
        )
        os.replace(tmp, DATA_FILE)   # 原子替换
    except OSError as e:
        raise StorageError(f"保存失败: {e}") from e
```
